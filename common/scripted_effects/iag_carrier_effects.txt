# Country scope. Spawns one player-owned carrier enclave system connected
# directly to the current capital, with two 25-size Gaia carrier colonies.
#
# The enclave is not meant for defense, resources, expansion, or normal
# automation. It is a stable UI command carrier for an external packet-rewrite
# executor.
#
# Access boundary: current vanilla script supports no verified pre-entry
# per-system allowlist for fleets. Grey Wind has one direct capital hyperlane
# and a player-owned outpost. A reactive on_entering_system_fleet hook detects
# intruders after arrival and sends eligible non-owner fleets MIA.

# Planet scope with a country root. Initializes either carrier world with the
# same neutral colony state and permanent safeguards. Role flags and visible
# names are assigned by the caller.
iag_initialize_carrier_enclave_planet_effect = {
    set_surveyed = {
        surveyed = yes
        surveyor = root
    }
    prevent_anomaly = yes
    clear_blockers = yes
    create_colony = {
        owner = root
    }
    set_controller = root
    create_pop_group = {
        size = 1000
        species = owner_main_species
    }
    set_colony_type = col_city
    iag_maintain_carrier_enclave_planet_effect = yes
}

iag_spawn_carrier_enclave_effect = {
    if = {
        limit = {
            NOT = {
                any_owned_planet = {
                    OR = {
                        has_carrier_flag = iag_carrier_enclave
                        solar_system = { has_star_flag = iag_carrier_enclave_system }
                    }
                }
            }
        }
        log = "[LLM Carrier] Carrier enclave spawn requested for [This.GetName]."

        capital_scope = {
            solar_system = {
                spawn_system = {
                    min_distance = 20
                    max_distance = 45
                    max_jumps = 0
                    initializer = iag_carrier_enclave_system
                    hyperlane = yes
                    is_discovered = yes
                    effect = {
                        save_event_target_as = iag_carrier_enclave_system
                        set_star_flag = iag_carrier_enclave_system
                        set_surveyed = {
                            surveyed = yes
                            surveyor = root
                        }
                    }
                }

                # Vanilla normally creates this lane because hyperlane = yes.
                # Keep the explicit fallback used by vanilla precursor events
                # so Grey Wind is guaranteed to connect to this capital system.
                if = {
                    limit = {
                        last_created_system = {
                            NOT = { has_hyperlane_to = prev }
                        }
                    }
                    add_hyperlane = {
                        from = this
                        to = last_created_system
                    }
                }

                last_created_system = {
                    save_event_target_as = iag_carrier_enclave_system
                    create_starbase = {
                        size = starbase_outpost
                        owner = root
                    }

                    every_system_planet = {
                        limit = { has_carrier_flag = iag_carrier_building_seed }
                        set_carrier_flag = iag_carrier_building_world
                        iag_initialize_carrier_enclave_planet_effect = yes
                        # create_colony may assign a generated colony name; set the
                        # visible name again after colony creation.
                        set_name = "NAME_IAG_Carrier_Enclave"
                        log = "[LLM Carrier] Building carrier Trantor initialized on [This.GetName]."
                    }

                    every_system_planet = {
                        limit = { has_carrier_flag = iag_carrier_zone_seed }
                        set_carrier_flag = iag_carrier_zone_world
                        iag_initialize_carrier_enclave_planet_effect = yes
                        set_name = "NAME_IAG_Carrier_Zone_Enclave"
                        log = "[LLM Carrier] District/zone carrier Terminus initialized on [This.GetName]."
                    }
                }
            }
        }

        set_country_flag = iag_carrier_enclave_created
        log = "[LLM Carrier] Carrier enclave spawn script completed for [This.GetName]."
    }
    else = {
        log = "[LLM Carrier] Carrier enclave spawn skipped for [This.GetName]: an owned carrier enclave already exists. Running repair instead."
        iag_repair_carrier_enclave_effect = yes
    }
}

# Country scope. Connects an existing Grey Wind system directly to the current
# capital. This is idempotent and is used for old saves created before 0.5.5.
iag_connect_carrier_to_capital_effect = {
    if = {
        limit = {
            exists = capital_scope
            any_owned_planet = {
                solar_system = {
                    has_star_flag = iag_carrier_enclave_system
                }
            }
        }

        capital_scope = {
            solar_system = {
                save_event_target_as = iag_carrier_capital_system
            }
        }

        random_system = {
            limit = {
                has_star_flag = iag_carrier_enclave_system
            }
            if = {
                limit = {
                    NOT = {
                        has_hyperlane_to = event_target:iag_carrier_capital_system
                    }
                }
                add_hyperlane = {
                    from = this
                    to = event_target:iag_carrier_capital_system
                }
                log = "[LLM Carrier] Grey Wind connected directly to the current capital system."
            }
        }
    }
}

# Country scope. Repairs role markers and safeguards. In pre-Terminus saves it
# also creates the missing district/zone carrier in the existing Grey Wind
# system. spawn_planet follows the vanilla Stellaris 4.4 Gaia-world pattern.
iag_repair_carrier_enclave_effect = {
    # New initializers retain role seed flags after colony creation.
    every_owned_planet = {
        limit = { has_carrier_flag = iag_carrier_building_seed }
        set_carrier_flag = iag_carrier_building_world
    }

    every_owned_planet = {
        limit = { has_carrier_flag = iag_carrier_zone_seed }
        set_carrier_flag = iag_carrier_zone_world
    }

    # Old saves have one generic carrier world. Promote it to Trantor before
    # creating Terminus.
    if = {
        limit = {
            NOT = {
                any_owned_planet = {
                    has_carrier_flag = iag_carrier_building_world
                }
            }
        }
        random_owned_planet = {
            limit = {
                OR = {
                    has_carrier_flag = iag_carrier_enclave
                    solar_system = { has_star_flag = iag_carrier_enclave_system }
                }
                NOT = { has_carrier_flag = iag_carrier_zone_world }
            }
            set_carrier_flag = iag_carrier_building_world
            log = "[LLM Carrier] Existing carrier promoted to the Trantor building role."
        }
    }

    # Upgrade an existing one-world enclave without creating another system.
    if = {
        limit = {
            any_owned_planet = {
                has_carrier_flag = iag_carrier_building_world
            }
            NOT = {
                any_owned_planet = {
                    has_carrier_flag = iag_carrier_zone_world
                }
            }
        }
        random_owned_planet = {
            limit = {
                has_carrier_flag = iag_carrier_building_world
            }
            solar_system = {
                spawn_planet = {
                    check_overlap = yes
                    class = pc_gaia
                    location = none
                    orbit_distance = { min = 90 max = 180 }
                    orbit_angle = { min = 0 max = 360 }
                    size = 25
                    flags = {
                        iag_carrier_enclave_seed
                        iag_carrier_zone_seed
                        iag_carrier_enclave
                    }
                    init_effect = {
                        set_carrier_flag = iag_carrier_zone_world
                        iag_initialize_carrier_enclave_planet_effect = yes
                        set_name = "NAME_IAG_Carrier_Zone_Enclave"
                        log = "[LLM Carrier] Missing Terminus carrier world created on [This.GetName]."
                    }
                }
            }
        }
    }

    every_owned_planet = {
        limit = {
            OR = {
                has_carrier_flag = iag_carrier_enclave
                solar_system = { has_star_flag = iag_carrier_enclave_system }
            }
        }
        iag_maintain_carrier_enclave_planet_effect = yes
    }

    every_owned_planet = {
        limit = { has_carrier_flag = iag_carrier_building_world }
        set_name = "NAME_IAG_Carrier_Enclave"
        log = "[LLM Carrier] Trantor building carrier repaired on [This.GetName]."
    }

    every_owned_planet = {
        limit = { has_carrier_flag = iag_carrier_zone_world }
        set_name = "NAME_IAG_Carrier_Zone_Enclave"
        log = "[LLM Carrier] Terminus district/zone carrier repaired on [This.GetName]."
    }

    iag_connect_carrier_to_capital_effect = yes
}

# Colony/carrier scope. Applies the carrier marker and permanent safeguard
# modifier. It also explicitly removes Fallen Empire holy-world markers. This
# planet is a player-owned command carrier, not a holy world; holy-world
# punishment in vanilla checks the holy_planet modifier, while other helpers
# check the holy_world_1..4 carrier flags.
iag_maintain_carrier_enclave_planet_effect = {
    # Stellaris 4.4 exposes many owned-colony loops as colony/carrier scope.
    # Use carrier flags here; planet flags in this scope log wrong-scope errors.
    set_carrier_flag = iag_carrier_enclave

    if = { limit = { has_modifier = holy_planet } remove_modifier = holy_planet }

    if = { limit = { has_carrier_flag = holy_world_1 } remove_carrier_flag = holy_world_1 }
    if = { limit = { has_carrier_flag = holy_world_2 } remove_carrier_flag = holy_world_2 }
    if = { limit = { has_carrier_flag = holy_world_3 } remove_carrier_flag = holy_world_3 }
    if = { limit = { has_carrier_flag = holy_world_4 } remove_carrier_flag = holy_world_4 }

    if = {
        limit = {
            NOT = { has_modifier = iag_carrier_enclave_planet }
        }
        add_modifier = {
            modifier = iag_carrier_enclave_planet
            days = -1
        }
    }
}
